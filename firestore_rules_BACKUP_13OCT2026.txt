rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAdmin() {
      return request.auth != null && request.auth.token.email == "admin@reward.rw";
    }
    function isOwner(userId) {
      return request.auth != null && request.auth.uid == userId;
    }
    function notTouchingProtected() {
      return !request.resource.data.diff(resource.data).affectedKeys().hasAny(['isBanned','isDeactivated','bannedBy','bannedAt','email','role','isAdmin']);
    }

    // USERS - allow postback to update balance
    match /users/{userId} {
      allow read: if isOwner(userId) || isAdmin();
      allow create: if isOwner(userId) || true;
      // Allow +10000 for postback (no auth), keep +20 limit for normal users
      allow update: if isAdmin() || (isOwner(userId) && notTouchingProtected() && 
        (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['balance']) || 
         request.resource.data.balance <= resource.data.balance + 20)
      ) || (request.auth == null && request.resource.data.diff(resource.data).affectedKeys().hasOnly(['balance']));
      allow delete: if isAdmin();

      match /referralDailyLogs/{docId} {
        allow read, write: if isOwner(userId) || isAdmin();
      }
      match /earnHistory/{docId} {
        allow read, write: if isOwner(userId) || isAdmin();
      }
      match /{sub=**} {
        allow read, write: if isOwner(userId) || isAdmin();
      }
    }

    // ADD THIS - For CPX postback
    match /transactions/{docId} {
      allow create: if true;
      allow read: if true;
      allow update, delete: if isAdmin();
    }

    match /usernames/{docId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if isAdmin();
    }

    match /deviceIds/{docId} {
      allow read: if true;
      allow create: if request.auth != null;
      allow update: if request.auth != null;
      allow delete: if isAdmin();
    }

    match /withdrawals/{docId} {
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow read: if isAdmin() || (request.auth != null && resource.data.userId == request.auth.uid);
      allow update, delete: if isAdmin();
    }

    match /supportTickets/{docId} {
      allow create: if request.auth != null;
      allow read: if isAdmin() || (request.auth != null && (resource.data.userId == request.auth.uid || resource.data.uid == request.auth.uid));
      allow update, delete: if isAdmin();
    }

    match /notifications/{docId} {
      allow read: if request.auth != null;
      allow create, delete: if isAdmin();
      allow update: if isAdmin();
    }

    match /adminLogs/{docId} { allow read, write: if isAdmin(); }
    match /appConfig/{docId} { allow read: if true; allow write: if isAdmin(); }
  }
}
